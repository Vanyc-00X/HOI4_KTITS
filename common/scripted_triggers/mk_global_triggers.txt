mk_has_faction = {
	is_in_faction = yes
	is_mk_country = yes
}

mk_can_form_ideology_faction = {
	is_mk_country = yes
	is_in_faction = no
	is_subject = no
	has_war = no
}

mk_can_accuse_revisionism = {
	is_mk_country = yes
	is_in_faction = yes
	is_subject = no
	NOT = { has_country_flag = mk_revisionist_cooldown }
	NOT = { has_idea = mk_idea_revisionist }
}

mk_can_condemn_revisionism = {
	is_mk_country = yes
	is_in_faction = yes
	NOT = { has_country_flag = mk_condemned_revisionism }
	any_other_country = {
		has_idea = mk_idea_revisionist
	}
}

mk_can_call_world_congress = {
	is_mk_country = yes
	check_variable = { mk_world_votes > 2 }
	NOT = { has_global_flag = mk_congress_cooldown }
	NOT = { has_global_flag = mk_congress_active }
}

mk_absurd_high = {
	check_variable = { global.mk_absurd_temp > 59 }
}

mk_absurd_critical = {
	check_variable = { global.mk_absurd_temp > 84 }
}

mk_great_turn_unlocked = {
	OR = {
		has_global_flag = mk_great_political_turn
		check_variable = { global.mk_absurd_temp > 39 }
		date > 1936.3.11
	}
}

mk_can_use_npt = {
	is_mk_country = yes
	has_government = fascism
}

mk_can_form_npt_protocol = {
	is_mk_country = yes
	is_in_faction = no
	is_subject = no
	has_war = no
	has_government = fascism
	check_variable = { mk_npt > 4 }
}

mk_can_mediate = {
	is_mk_country = yes
	has_war = no
	OR = {
		original_tag = ADL
		original_tag = KMS
		check_variable = { mk_world_votes > 2 }
	}
}

mk_era_forks = {

}

}

mk_can_win_ideological = {

}

mk_can_win_economic = {
	is_mk_country = yes

}

mk_can_win_npt = {
	is_mk_country = yes
	has_government = fascism
	NOT = { has_country_flag = mk_victory_claimed }
	check_variable = { mk_npt > 7 }
	check_variable = { global.mk_congress_count > 2 }
	check_variable = { global.mk_absurd_temp > 49 }
}

	is_mk_country = yes
	NOT = { has_country_flag = mk_victory_claimed }
	mk_absurd_critical = yes
	has_war = yes
}

mk_can_win_integration = {
	is_mk_country = yes
	NOT = { has_country_flag = mk_victory_claimed }
