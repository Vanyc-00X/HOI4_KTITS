		limit = { NOT = { has_global_flag = mk_globals_initialized } }

		set_variable = { global.mk_congress_count = 0 }
		set_variable = { global.mk_revisionism_cases = 0 }
		set_variable = { global.mk_condemn_votes = 0 }

		set_variable = { global.mk_congress_no_votes = 0 }
		set_variable = { global.mk_congress_total_votes = 0 }
		set_variable = { global.mk_congress_required_votes = 0 }
		set_variable = { global.mk_same_ideology_count = 1 }
		clr_global_flag = mk_congress_active
		clr_global_flag = mk_congress_cooldown
		clr_global_flag = mk_great_political_turn
		every_country = {
			limit = { is_mk_country = yes }
			mk_init_country_mechanics = yes
		}
		news_event = { id = mk_global.1 hours = 2 }
		news_event = { id = mk_plot.1 days = 30 }
		news_event = { id = mk_flavor.6 days = 55 }
		every_country = {
			limit = { is_mk_country = yes }
			country_event = { id = mk_plot.2 days = 32 }
			country_event = { id = mk_flavor.2 days = 45 }
		}
	}
}

mk_init_country_mechanics = {
	set_variable = { mk_world_votes = 1 }
	set_variable = { mk_npt = 0 }
	set_variable = { mk_revisionism = 0 }
	set_variable = { mk_misconduct = 0 }
	clr_country_flag = mk_congress_vote_yes
	clr_country_flag = mk_congress_vote_no
	clr_country_flag = mk_congress_target
	clr_country_flag = mk_congress_attacker
	clr_country_flag = mk_ruler_exiled
	clr_country_flag = mk_revisionist_cooldown
	clr_country_flag = mk_npt_surge
	clr_country_flag = mk_condemned_revisionism
	clr_country_flag = mk_victory_claimed
	add_ideas = mk_idea_world_vote
	mk_mech_init = yes
	if = {
		limit = { original_tag = VCI }
		set_variable = { mk_world_votes = 3 }
		add_ideas = mk_idea_vci_host
	}
	# BTR and CRE begin with a modest misconduct record; declared wars increase it.
	if = {
		limit = {
			OR = {
				original_tag = BTR
				original_tag = CRE
			}
		}
		set_variable = { mk_misconduct = 1 }
	}
	# Stage 11: NPT seed only for fascist starting governments
	if = {
		limit = {
			OR = {
				original_tag = BTR
				original_tag = CRE
			}
		}
		set_variable = { mk_npt = 2 }
		add_ideas = mk_idea_npt_seed
	}
}

# Start one weighted global vote. The sponsor's three (or two emergency) votes are committed up front.
mk_start_congress_vote = {
	every_country = {
		clr_country_flag = mk_congress_target
		clr_country_flag = mk_congress_vote_yes
		clr_country_flag = mk_congress_vote_no
	}
	set_variable = { global.mk_congress_yes_votes = mk_congress_sponsor_weight }
	set_variable = { global.mk_congress_no_votes = 0 }
	set_variable = { global.mk_congress_total_votes = 0 }
	set_variable = { global.mk_congress_required_votes = 0 }
	save_global_event_target_as = mk_congress_caller
	random_other_country = {
		limit = {
			is_mk_country = yes
			is_subject = no
			NOT = { has_capitulated = yes }
			check_variable = { mk_misconduct > 0 }
		}
		set_country_flag = mk_congress_target
		save_global_event_target_as = mk_congress_target
	}
	# Only sovereign, non-capitulated MK states vote; abstentions count toward the quorum.
	every_country = {
		limit = {
			is_mk_country = yes
			is_subject = no
			NOT = { has_capitulated = yes }
			NOT = { has_country_flag = mk_congress_target }
		}
		add_to_variable = { global.mk_congress_total_votes = mk_world_votes }
	}
	# The sponsor already spent these points to call the session; count its committed ballot once.
	add_to_variable = { global.mk_congress_total_votes = mk_congress_sponsor_weight }
	set_variable = { global.mk_congress_required_votes = global.mk_congress_total_votes }
	divide_variable = { var = global.mk_congress_required_votes value = 2 }
	set_country_flag = mk_congress_vote_yes
	every_other_country = {
		limit = {
			is_mk_country = yes
			is_subject = no
			NOT = { has_capitulated = yes }
			NOT = { has_country_flag = mk_congress_target }
		}
		country_event = { id = mk_global.10 hours = 1 random_hours = 8 }
	}
	event_target:mk_congress_caller = {
		country_event = { id = mk_global.11 days = 7 }
	}
}

mk_add_absurd = {
	add_to_variable = { global.mk_absurd_temp = mk_absurd_delta }
	clamp_variable = {
		var = global.mk_absurd_temp
		min = 0
		max = 100
	}
	mk_refresh_absurd_ideas = yes
	if = {
		limit = {
			check_variable = { global.mk_absurd_temp > 39 }
			NOT = { has_global_flag = mk_great_political_turn }
		}
		mk_fire_great_turn = yes
	}
	if = {
		limit = {
			check_variable = { global.mk_absurd_temp > 84 }
			NOT = { has_global_flag = mk_absurd_critical_fired }
		}
		set_global_flag = mk_absurd_critical_fired
		news_event = { id = mk_global.5 hours = 1 }
	}
}

mk_sub_absurd = {
	subtract_from_variable = { global.mk_absurd_temp = mk_absurd_delta }
	clamp_variable = {
		var = global.mk_absurd_temp
		min = 0
		max = 100
	}
	mk_refresh_absurd_ideas = yes
}

mk_congress_deescalation_effect = {
	set_temp_variable = { mk_votes_delta = 2 }
	mk_spend_votes = yes
	set_temp_variable = { mk_absurd_delta = 8 }
	mk_sub_absurd = yes
	add_to_variable = { global.mk_congress_count = 1 }
	every_country = {
		limit = { is_mk_country = yes }
		add_stability = 0.01
		add_timed_idea = { idea = mk_idea_congress_mandate days = 90 }
		if = {
			limit = { check_variable = { mk_misconduct > 0 } }
			subtract_from_variable = { mk_misconduct = 1 }
			clamp_variable = { var = mk_misconduct min = 0 max = 20 }
		}
	}
}

mk_refresh_absurd_ideas = {
	every_country = {
		limit = { is_mk_country = yes }
		remove_ideas = mk_idea_absurd_calm
		remove_ideas = mk_idea_absurd_warm
		remove_ideas = mk_idea_absurd_hot
		remove_ideas = mk_idea_absurd_critical
		if = {
			limit = { check_variable = { global.mk_absurd_temp < 25 } }
			add_ideas = mk_idea_absurd_calm
		}
		else_if = {
			limit = { check_variable = { global.mk_absurd_temp < 50 } }
			add_ideas = mk_idea_absurd_warm
		}
		else_if = {
			limit = { check_variable = { global.mk_absurd_temp < 75 } }
			add_ideas = mk_idea_absurd_hot
		}
		else = {
			add_ideas = mk_idea_absurd_critical
		}
	}
}

mk_monthly_absurd_tick = {
	if = {
		limit = { has_global_flag = mk_globals_initialized }
		set_temp_variable = { mk_absurd_delta = 1 }
		if = {
			limit = { has_global_flag = mk_congress_active }
			set_temp_variable = { mk_absurd_delta = 2 }
		}
		if = {
			limit = { check_variable = { global.mk_revisionism_cases > 2 } }
			add_to_temp_variable = { mk_absurd_delta = 1 }
		}
		mk_add_absurd = yes
		mk_count_same_ideology = yes
		mk_fire_timeline_events = yes
	}
}

# Accuser leaves the alliance and gains wargoals on all remaining members (TZ §5.1)
mk_accuse_revisionism_effect = {
	add_to_variable = { mk_revisionism = 1 }
	add_to_variable = { global.mk_revisionism_cases = 1 }
	set_variable = { global.mk_condemn_votes = 0 }
	set_country_flag = { flag = mk_revisionist_cooldown days = 180 }
	add_timed_idea = { idea = mk_idea_revisionist days = 180 }
	add_political_power = 40
	add_war_support = 0.08
	set_temp_variable = { mk_absurd_delta = 10 }
	mk_add_absurd = yes
	every_country = {
		limit = {
			is_in_faction_with = ROOT
			NOT = { tag = ROOT }
		}
		ROOT = {
			create_wargoal = {
				type = mk_revisionist_aggression
				target = PREV
			}
		}
		create_wargoal = {
			type = mk_revisionist_aggression
			target = ROOT
		}
		add_timed_idea = { idea = mk_idea_revision_hesitation days = 100 }
		clr_country_flag = mk_condemned_revisionism
		country_event = { id = mk_global.23 hours = 6 }
	}
	leave_faction = yes
	remove_ideas = mk_idea_faction_member
	news_event = { id = mk_global.22 hours = 2 }
	country_event = { id = mk_global.20 hours = 4 }
}

mk_condemn_revisionism_effect = {
	set_country_flag = mk_condemned_revisionism
	add_to_variable = { global.mk_condemn_votes = 1 }
	add_political_power = 20
	add_war_support = 0.05
	add_timed_idea = { idea = mk_idea_condemn_revisionism days = 90 }
	if = {
		limit = { check_variable = { global.mk_condemn_votes > 2 } }
		if = {
			limit = { NOT = { has_global_flag = mk_condemn_majority } }
			set_global_flag = mk_condemn_majority
			every_country = {
				limit = {
					is_mk_country = yes
					is_in_faction = yes
					has_country_flag = mk_condemned_revisionism
				}
				add_political_power = 30
				add_stability = 0.03
				country_event = { id = mk_global.24 hours = 4 }
			}
		}
	}
}

mk_form_ideology_faction_effect = {
	if = {
		limit = { has_government = democratic }
		create_faction = MK_FACTION_LIBERAL
	}
	else_if = {
		limit = { has_government = communism }
		create_faction = MK_FACTION_COLLECTIVE
	}
	else_if = {
		limit = { has_government = fascism }
		create_faction = MK_FACTION_ORDER
	}
	else = {
		create_faction = MK_FACTION_NEUTRAL
	}
	add_ideas = mk_idea_faction_member
	add_political_power = -50
	set_temp_variable = { mk_absurd_delta = 3 }
	mk_add_absurd = yes
	# Stage 13: instantly invite every peaceful same-ideology MK country
	if = {
		limit = { has_war = no }
		every_other_country = {
			limit = {
				is_mk_country = yes
				has_government = ROOT
				is_in_faction = no
				is_subject = no
				has_war = no
			}
			ROOT = { add_to_faction = PREV }
			add_ideas = mk_idea_faction_member
			country_event = { id = mk_global.30 hours = 4 }
		}
	}
	news_event = { id = mk_global.3 hours = 4 }
	news_event = { id = mk_flavor.1 hours = 8 }
}
